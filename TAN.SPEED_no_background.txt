local Bypass = {
    enabled        = false,
    power          = 97000,
    mode           = "PC",             -- "PC" | "MOBILE"
    version        = "V1",             -- "V1" | "V2"
    autoOnSteal    = false,
    keybind        = Enum.KeyCode.E,
    spamThread     = nil,
    payload        = nil,
    lastStealState = false,
}

-- Puissances par défaut selon le mode
local DEFAULT_PC_POWER     = 97000
local DEFAULT_MOBILE_POWER = 70000

-- ---------------------------------------------------------------------
-- Construction des payloads (identiques à Bubble Hub)
-- ---------------------------------------------------------------------
local function buildPayloadV1(power)
    local root = {{}}
    local cur  = root[1]
    for _ = 1, 186 do
        local t = {}
        table.insert(cur, t)
        cur = t
    end
    local list = {}
    local count = math.floor(power / 188)
    for i = 1, count do
        list[i] = root
    end
    return list
end

local function buildPayloadV2(power)
    local root = {{}}
    local cur  = root[1]
    for _ = 1, 296 do
        local t = {}
        table.insert(cur, t)
        cur = t
    end
    local list = {}
    local count = math.floor(power / 298)
    for i = 1, count do
        list[i] = root
    end
    return list
end

local function buildPayload(power)
    if Bypass.version == "V1" then
        return buildPayloadV1(power)
    end
    return buildPayloadV2(power)
end

local function getBlockRemote()
    local rrs = game:FindFirstChild("RobloxReplicatedStorage")
    return rrs and rrs:FindFirstChild("SetPlayerBlockList")
end

-- ---------------------------------------------------------------------
-- Start / Stop du spam
-- ---------------------------------------------------------------------
local function startBypass()
    if Bypass.enabled then return end
    local remote = getBlockRemote()
    if not remote then
        warn("[TAN.SPEED] Remote 'SetPlayerBlockList' introuvable.")
        return
    end

    pcall(function() NetworkClient:SetOutgoingKBPSLimit(math.huge) end)

    Bypass.enabled = true
    Bypass.payload = buildPayload(Bypass.power)
    Bypass.spamThread = task.spawn(function()
        while Bypass.enabled do
            if Bypass.payload then
                pcall(function() remote:FireServer(Bypass.payload) end)
            end
            task.wait(0.12)
        end
    end)
end

local function stopBypass()
    Bypass.enabled = false
    if Bypass.spamThread then
        pcall(task.cancel, Bypass.spamThread)
        Bypass.spamThread = nil
    end
    Bypass.payload = nil
    pcall(function() NetworkClient:SetOutgoingKBPSLimit(math.huge) end)
end

-- =====================================================================
-- Helpers UI
-- =====================================================================
local function tween(obj, props, t)
    TweenService:Create(
        obj,
        TweenInfo.new(t or 0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        props
    ):Play()
end

local function refreshPowerDisplay()
    local n = Bypass.power
    TextButton_2.Text = (n >= 1000) and (math.floor(n / 1000) .. "K") or tostring(n)
    TextBox.Text = tostring(n)
end

local function setPower(power)
    power = tonumber(power)
    if not power then return end
    local maxP = Bypass.mode == "PC" and 150000 or 100000
    power = math.clamp(math.floor(power), 10000, maxP)
    Bypass.power = power
    refreshPowerDisplay()
    if Bypass.enabled then
        Bypass.payload = buildPayload(Bypass.power)
    end
end

local function updateActivatorVisual()
    local active = Bypass.enabled
    TextButton_7.Text = active and "DEACTIVATE" or "ACTIVATE"
    tween(TextButton_7, {
        BackgroundColor3 = active and Color3.fromRGB(165, 28, 36) or Color3.fromRGB(80, 120, 255),
    })
    local ringColor = active and Color3.fromRGB(60, 220, 120) or Color3.fromRGB(80, 120, 255)
    tween(UIStroke_3, { Color = ringColor, Transparency = active and 0.10 or 0.55 })
    tween(UIStroke_4, { Color = ringColor, Transparency = active and 0.10 or 0.75 })
end

local function setBypass(on)
    if on then startBypass() else stopBypass() end
    updateActivatorVisual()
end

-- Déplacement via l'en-tête
do
    local header = Frame
    header.Active = true
    local dragging, dragStart, startPos = false, nil, nil

    header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging   = true
            dragStart  = input.Position
            startPos   = Main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging
        and (input.UserInputType == Enum.UserInputType.MouseMovement
             or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

-- PC / MOBILE : change le mode + force la puissance (97K PC / 70K MOBILE)
do
    local function setPlatform(p)
        Bypass.mode = p
        local isPC = (p == "PC")

        -- Couleurs des boutons
        tween(TextButton_3, {
            BackgroundTransparency = isPC and 0 or 1,
            TextColor3 = isPC and Color3.fromRGB(245, 250, 255) or Color3.fromRGB(80, 100, 180),
        })
        tween(TextButton_4, {
            BackgroundTransparency = isPC and 1 or 0,
            BackgroundColor3     = Color3.fromRGB(80, 120, 255),
            TextColor3           = isPC and Color3.fromRGB(80, 100, 180) or Color3.fromRGB(245, 250, 255),
        })

        -- ⚡ Puissance automatique : 97K en PC / 70K en MOBILE
        setPower(isPC and DEFAULT_PC_POWER or DEFAULT_MOBILE_POWER)
    end
    TextButton_3.MouseButton1Click:Connect(function() setPlatform("PC")     end)
    TextButton_4.MouseButton1Click:Connect(function() setPlatform("MOBILE") end)
    setPlatform("PC")
end

-- Keybind : clic → attend une touche, puis l'utilise pour toggle le bypass
do
    local listening = false
    TextButton_6.MouseButton1Click:Connect(function()
        listening = true
        TextLabel_6.Text = "..."
    end)

    UserInputService.InputBegan:Connect(function(input, gameProcessed)
        if listening and input.UserInputType == Enum.UserInputType.Keyboard then
            Bypass.keybind  = input.KeyCode
            TextLabel_6.Text = input.KeyCode.Name
            listening = false
            return
        end
        if gameProcessed then return end
        if input.UserInputType == Enum.UserInputType.Keyboard
        and input.KeyCode == Bypass.keybind then
            setBypass(not Bypass.enabled)
        end
    end)
end

-- ACTIVATE : bascule le bypass
TextButton_7.MouseButton1Click:Connect(function()
    setBypass(not Bypass.enabled)
end)

local function hover(btn, over, normal)
    btn.MouseEnter:Connect(function() tween(btn, { BackgroundTransparency = over   }, 0.12) end)
    btn.MouseLeave:Connect(function() tween(btn, { BackgroundTransparency = normal }, 0.12) end)
end

task.spawn(function()
    while TANSPEEDGui.Parent do
        local stealing = LocalPlayer:GetAttribute("Stealing") == true
        if stealing ~= Bypass.lastStealState then
            Bypass.lastStealState = stealing
            if Bypass.autoOnSteal and stealing and not Bypass.enabled then
                setBypass(true)
            end
        end
        task.wait(0.1)
    end
end)
